# .gitignore Cleanup Commands
# ============================
# These commands will remove files from Git tracking that should be ignored.
# Run these commands from your repository root in PowerShell or Git Bash.

# IMPORTANT: Make sure you've committed or backed up any important work first!

# Navigate to repository root
cd "C:\Users\chibl\OneDrive\Documents\Coding Projects\Stage of Dreams"

# ============================================
# Step 1: Remove Unity-generated folders from tracking
# ============================================
git rm -r --cached Library/
git rm -r --cached Temp/
git rm -r --cached obj/
git rm -r --cached Logs/
git rm -r --cached UserSettings/

# ============================================
# Step 2: Remove auto-generated Visual Studio files
# ============================================
git rm --cached *.csproj
git rm --cached *.sln
git rm --cached "Assembly-CSharp.csproj"
git rm --cached "Assembly-CSharp-Editor.csproj"
git rm --cached "NewAssembly.csproj"
git rm --cached "Scripts.csproj"
git rm --cached "Stage of Dreams.sln"

# ============================================
# Step 3: Remove Unity meta files from Docs folder (optional)
# ============================================
# Unity creates .meta files for folders outside Assets/
# These are usually not needed in version control
git rm --cached "Docs/Examples and Guides/Dialog System Guide.md.meta"
git rm --cached "Docs/Examples and Guides/DialogNavigationExample.cs.meta"
git rm --cached "Docs/Examples and Guides/DialogNodeExamples.cs.meta"
git rm --cached "Docs/Examples and Guides/IntegratedDialogSystemExample.cs.meta"
git rm --cached "Docs/Examples and Guides/UIToolkitDialogSetup.cs.meta"

# ============================================
# Step 4: Remove UIElementsSchema (Unity auto-generated)
# ============================================
git rm -r --cached UIElementsSchema/

# ============================================
# Step 5: Commit the cleanup
# ============================================
git commit -m "chore: Remove ignored files from Git tracking

- Removed Unity-generated folders (Library, Temp, obj, Logs, UserSettings)
- Removed auto-generated .csproj and .sln files
- Removed Unity .meta files from Docs folder
- Removed UIElementsSchema auto-generated folder
- Fixed .gitignore location (moved from .github/ to root)

These files will be regenerated by Unity/Visual Studio as needed."

# ============================================
# Step 6: Verify cleanup worked
# ============================================
# Check that ignored files are no longer tracked
git status --short

# Check a specific file/folder to confirm it's ignored
git check-ignore -v Library/ Temp/ obj/ *.csproj *.sln

# ALTERNATIVE: One-line command to remove all at once
# ============================================

# NOTE: These files will still exist on your disk
# ============================================
# The --cached flag only removes files from Git tracking, not from your filesystem.
# This is what you want - Unity and Visual Studio need these files to work properly!
